# ===== Configuración del cliente cli-linux (192.168.1.30) =====

# /etc/defaultdomain
farmadrogas.local

# /etc/yp.conf
domain farmadrogas.local server 192.168.1.10

# /etc/nsswitch.conf (líneas modificadas: se agregó "nis" al final)
passwd:         files systemd sss nis
group:          files systemd sss nis
shadow:         files systemd sss nis
hosts:          files mdns4_minimal [NOTFOUND=return] dns nis

# /etc/systemd/system/systemd-logind.service.d/nis.conf
[Service]
IPAddressAllow=192.168.1.0/24

# /etc/fstab (líneas agregadas)
192.168.1.10:/srv/nfs/farmadrogas  /mnt/farmadrogas  nfs  defaults,_netdev  0  0
192.168.1.10:/srv/nfs/home         /srv/nfs/home     nfs  defaults,_netdev  0  0

# /etc/chrony/chrony.conf (fuentes de internet comentadas; línea agregada al final)
server 192.168.1.10 iburst prefer
